<script>
let moviesData = [];
let currentPage = 1;

async function getMovies(page = 1) {

    const genre = document.getElementById("genre").value;
    const industry = document.getElementById("industry").value;
    const era = document.getElementById("era").value;

    let start, end;

    if (era === "1980") {
        start = "1980-01-01"; end = "1989-12-31";
    } else if (era === "1990") {
        start = "1990-01-01"; end = "1999-12-31";
    } else if (era === "2000") {
        start = "2000-01-01"; end = "2009-12-31";
    } else if (era === "2010") {
        start = "2010-01-01"; end = "2019-12-31";
    } else {
        start = "2020-01-01"; end = "2030-12-31";
    }

    const url =
    `https://api.themoviedb.org/3/discover/movie?api_key=${API_KEY}` +
    `&with_genres=${genre}` +
    `&with_original_language=${industry}` +
    `&primary_release_date.gte=${start}` +
    `&primary_release_date.lte=${end}` +
    `&sort_by=popularity.desc` +
    `&page=${page}`;

    document.getElementById("movies").innerHTML = "Loading...";

    const res = await fetch(url);
    const data = await res.json();

    moviesData = data.results;
    currentPage = page;

    let output = "";

    data.results.forEach((movie, index) => {

        output += `
        <div class="card" onclick="showMovie(${index})">
            <img src="https://image.tmdb.org/t/p/w500${movie.poster_path}">
            <div class="card-content">
                <h3>${movie.title}</h3>
                <p>⭐ ${movie.vote_average}</p>
            </div>
        </div>`;
    });

    document.getElementById("movies").innerHTML = output + pagination();
}

function showMovie(index) {

    const movie = moviesData[index];

    const popup = document.createElement("div");
    popup.className = "popup";

    popup.innerHTML = `
        <div class="popup-box">
            <h2>${movie.title}</h2>

            <p><b>Rating:</b> ⭐ ${movie.vote_average}</p>

            <p>${movie.overview}</p>

            <p id="ott-${movie.id}">Loading OTT platforms...</p>

            <button onclick="watchTrailer(${movie.id})">
                ▶ Watch Trailer
            </button>

            <br><br>

            <button onclick="this.parentElement.parentElement.remove()">
                Close
            </button>
        </div>
    `;

    document.body.appendChild(popup);

    getOTT(movie.id);
}

async function getOTT(movieId) {

    try {

        const response = await fetch(
            `https://api.themoviedb.org/3/movie/${movieId}/watch/providers?api_key=${API_KEY}`
        );

        const data = await response.json();

        const ottElement = document.getElementById(`ott-${movieId}`);

        if (!ottElement) return;

        const region =
            data.results?.IN ||
            data.results?.US ||
            Object.values(data.results || {})[0];

        if (region && region.flatrate) {

            let providers = "<b>Available on:</b><br>";

            region.flatrate.forEach(provider => {
                providers += provider.provider_name + "<br>";
            });

            ottElement.innerHTML = providers;

        } else {

            ottElement.innerHTML = "OTT information not available";

        }

    } catch (error) {

        console.error(error);

        document.getElementById(`ott-${movieId}`).innerHTML =
            "Error loading OTT information";
    }
}

async function watchTrailer(movieId) {

    try {

        const response = await fetch(
            `https://api.themoviedb.org/3/movie/${movieId}/videos?api_key=${API_KEY}`
        );

        const data = await response.json();

        const trailer = data.results.find(
            video =>
                video.site === "YouTube" &&
                video.type === "Trailer"
        );

        if (trailer) {

            window.open(
                `https://www.youtube.com/watch?v=${trailer.key}`,
                "_blank"
            );

        } else {

            alert("Trailer not available");

        }

    } catch (error) {

        console.error(error);
        alert("Error loading trailer");

    }
}

function pagination() {
    return `
    <div class="pagination">
        <button onclick="getMovies(${currentPage - 1})" ${currentPage === 1 ? "disabled" : ""}>Prev</button>
        <span>Page ${currentPage}</span>
        <button onclick="getMovies(${currentPage + 1})">Next</button>
    </div>
    `;
}
